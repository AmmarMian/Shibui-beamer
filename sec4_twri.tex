\begin{frame}{Kronecker Dictionary Structure}

  \begin{columns}[T]
    \column{0.50\textwidth}
      \vspace{0.3em}

      \textbf{Propagation dictionary:}

      \vspace{0.2em}
      \begin{shibuiframe}{ }
        \small
        \[
          [\Mdict]_{:,p} = [e^{-j\pulsation_1 \tau_p}, \ldots, e^{-j\pulsation_{\Mnbfreq} \tau_p}]^{\T}
        \]
        \vspace{0.1em}
        \begin{itemize}
          \setlength{\itemsep}{0.06em}
          \item Column $p$: steering vector
          \item $\tau_p$: propagation delay
          \item From wall geometry + dielectric
        \end{itemize}
      \end{shibuiframe}

      \vspace{0.3em}
      \textbf{Kronecker structure:}
      \begin{itemize}
        \setlength{\itemsep}{0.06em}
        \item $\identmat{\Mnbpos} \kron \Vtarget$: same reflectivity per antenna
        \pause
        \item $\twoonenorm{\Vtarget}$: group sparsity per cell
        \pause
        \item Zero groups $\Rightarrow$ no target
      \end{itemize}

      \vspace{0.2em}
      \citebadge{Setlur et al. (2012)}

    \column{0.50\textwidth}
      \vspace{0.3em}
      \centering
      \figplaceholder[0.8\linewidth]{%
        Dictionary columns:\\steering vectors}
  \end{columns}
\end{frame}

% SLIDE 3 -- KRPCA: Problem Formulation
% ------------------------------------------------------------------
\begin{frame}{KRPCA: Problem Formulation}

  \begin{shibuiframe}{Kronecker Robust PCA (KRPCA)}
    \[
      \argminOn{\Mwall,\,\Vtarget}
        \nuclearnorm{\Mwall}
        + \sparsityhp\,\twoonenorm{\Vtarget}
      \quad
      \text{s.t.}
      \quad
      \Mtwri = \Mwall + \Mdict\bigl(\identmat{\Mnbpos} \kron \Vtarget\bigr)
    \]
  \end{shibuiframe}

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \vspace{0.2em}
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.08em}
        \item $\nuclearnorm{\Mwall}$: low-rank (coherent)
        \item $\twoonenorm{\Vtarget}$: group sparsity
        \item Constraint: exact fit (noise-free)
        \item ADMM solver: same as L2-GPR
      \end{itemize}

    \column{0.50\textwidth}
      \vspace{0.2em}
      \begin{shibuitip}[Same structure]
        \tiny
        \textbf{GPR}: $\min \nuclearnorm{\Mclutter} + \sparsityhp\sum_{\iatom}\lonenorm{\Mcoef_{\iatom}}$\\
        \textbf{TWRI}: $\min \nuclearnorm{\Mwall} + \sparsityhp\twoonenorm{\Vtarget}$
      \end{shibuitip}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
% SLIDE 4 -- HKRPCA: Huber Extension  [skippable]
% ------------------------------------------------------------------
\begin{frame}{HKRPCA: Huber Extension \skipbadge{}}

  \textbf{Adding robustness to non-Gaussian wall returns:}

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \vspace{0.2em}
      \textbf{Variant 1 --- Semi-splitting PGD:}

      \begin{shibuiframe}{ }
        \footnotesize
        \[
          \argminOn{\Mwall,\,\Vtarget}
            \hubernorm{\Mtwri - \Mwall
              - \Mdict(\identmat{\Mnbpos}\kron\Vtarget)}^{2}
            + \nuclearnorm{\Mwall}
            + \sparsityhp\,\twoonenorm{\Vtarget}
        \]
      \end{shibuiframe}

      \vspace{0.3em}
      \begin{itemize}
        \item Huber data fidelity replaces $\ell_2$
        \item Solved by \emphtext{proximal gradient descent} (PGD)
        \item Single noise model for whole measurement
      \end{itemize}

    \column{0.50\textwidth}
      \vspace{0.2em}
      \textbf{Variant 2 --- Full-splitting MM (column-wise):}

      \vspace{0.3em}
      \begin{itemize}
        \item Models noise \emphtext{column-by-column}\\
              (different noise per antenna/sensor)
        \item More robust to \emphtext{sensor-specific artefacts}
        \item Solved by \emphtext{majorization-minimization} (MM)\\
              with full variable splitting
      \end{itemize}
  \end{columns}

  \vspace{0.5em}
  \begin{shibuiproposition}[HKRPCA wall update]
    The wall component update admits a \emph{Huber proximal} closed form,
    analogous to Hub-GPR:
    \[
      \Mwall^{(\iadmm+1)}
        = \pxHub{1/\laghpclutter}{\delta}\!\bigl(
            \mathbf{V}_{L} + \mathbf{A}_{L}
          \bigr)
          - \mathbf{A}_{L},
    \]
    where
    $\mathbf{A}_{L}
       = \Mdict\bigl(\identmat{\Mnbpos}\kron\Vtarget^{(\iadmm+1)}\bigr)
         - \Mtwri$
    and
    $\mathbf{V}_{L} = \Mauxclutter + \Mdualclutter$.
  \end{shibuiproposition}

\end{frame}

  \vspace{0.5em}
  \textbf{Key takeaways:}

  \vspace{0.3em}
  \begin{itemize}
    \item TWRI problem $=$ GPR problem in \emphtext{frequency domain}
          with \emphtext{Kronecker dictionary} structure
    \item Huber norm extension provides robustness to
          \emphtext{non-Gaussian wall returns} and sensor artefacts
    \item HKRPCA achieves \emphtext{better target localization}
          under adverse conditions, especially the full-splitting variant
  \end{itemize}
