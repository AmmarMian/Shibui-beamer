\begin{frame}{Summary of Contributions}

  \begin{columns}[T]

    % ------- Block 1: GPR methods -------
    \column{0.33\textwidth}
      \begin{shibuiframe}{L2-GPR \& Hub-GPR}
        \footnotesize
        \begin{itemize}
          \item Conv.\ sparse + low-rank clutter
          \pause
          \item Physics dict: $\nbatoms = 25$ atoms
          \pause
          \item Huber: robust to non-Gaussian
          \pause
          \item Closed-form for update
          \item Applications: inspection, demining
        \end{itemize}
      \end{shibuiframe}

    % ------- Block 2: TWRI / HKRPCA -------
    \column{0.33\textwidth}
      \begin{shibuiframe}{TWRI \& HKRPCA}
        \footnotesize
        \begin{itemize}
          \item Kronecker dictionary structure
          \pause
          \item KRPCA + Huber variants
          \pause
          \item Group sparsity for localization
        \end{itemize}
      \end{shibuiframe}

    % ------- Block 3: LCRPCA deep unrolling -------
    \column{0.33\textwidth}
      \begin{shibuiframe}{LCRPCA (unrolling)}
        \footnotesize
        \begin{itemize}
          \item Unrolled PGD architecture
          \pause
          \item Learnable filters (robust)
          \pause
          \item $21{,}656$ vs $8.6\mathrm{M}$ params
          \pause
          \item PR-AUC $= 0.85$ (10\% data)
        \end{itemize}
      \end{shibuiframe}

  \end{columns}

  \vspace{0.6em}
  \begin{shibuiwarning}
    \textbf{Unifying principle:}
    all three methods exploit the same structural prior ---
    \emphtext{low-rank clutter} + \emphtext{sparse target signal}
    --- using ADMM and proximal operators.
  \end{shibuiwarning}

\end{frame}
