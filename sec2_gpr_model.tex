\begin{frame}{B-scan Decomposition}

  % Top: decomposition model in a highlighted box
  \begin{shibuiframe}{Signal model}
    \[
      \Mimage
        = \underbrace{%
            \sum_{\iatom=1}^{\nbatoms}
              \Mcoef_{\iatom} \Mconv \Matom_{\iatom}%
          }_{\text{target responses}}
        + \underbrace{\Mclutter}_{\text{clutter (low-rank)}}
        + \underbrace{\Mnoise}_{\text{noise}}
    \]
  \end{shibuiframe}

  \vspace{0.5em}

  % Below: two-column layout
  \begin{columns}[T]
    \column{0.50\textwidth}
      \vspace{0.3em}
      \begin{itemize}
        \item $\Matom_{\iatom}$: physics-based \emphtext{hyperbola atom} $k$,
              $k = 1,\ldots,\nbatoms$,\quad $\nbatoms = 25$
        \item $\Mcoef_{\iatom}$: \emphtext{sparse coefficient map}
              (target location \& strength)
        \item $\Mclutter$: ground/layer clutter,
              assumed \emphtext{low-rank}
        \item $\Mnoise$: additive measurement noise
      \end{itemize}

    \column{0.50\textwidth}
      \vspace{0.3em}
      \centering
      \figplaceholder[0.85\linewidth]{%

\end{frame}

\begin{frame}{Physics-Based Dictionary $\{\Matom_{\iatom}\}$}

  \begin{columns}[T]
    \column{0.52\textwidth}
      \vspace{0.3em}

      \textbf{Atom construction:}

      \vspace{0.2em}
      \begin{shibuiframe}{ }
        \small
        \[
          h_{k}(x,t) = r\!\left(t - \frac{2}{\speed}\sqrt{(x - x_0)^2 + d^2}\right)
        \]
        \vspace{0.1em}
        \begin{itemize}
          \setlength{\itemsep}{0.05em}
          \item $r(t)$: Ricker wavelet
          \item $\speed = \lightspeed/\sqrt{\permittivity}$: EM speed
          \item $d$: target depth
          \item $x_0$: horizontal position
        \end{itemize}
      \end{shibuiframe}

      \vspace{0.3em}
      \begin{itemize}
        \setlength{\itemsep}{0.08em}
        \item Each $\Matom_{\iatom}$ encodes one hyperbola for a specific depth/permittivity
        \item Dictionary: $\nbatoms = 25$ atoms varying in \emphtext{curvature}
      \end{itemize}

      \vspace{0.3em}
      \citebadge{Terrasse (2016) --- GPR dictionary}

    \column{0.48\textwidth}
      \vspace{0.3em}
      \centering
      \figplaceholder[0.85\linewidth]{%

        (hyperbola shapes, $2\times 2$ grid)}
  \end{columns}
\end{frame}

\begin{frame}{L2-GPR: Problem Formulation}

  \begin{shibuiframe}{Optimization problem}
    \small
    \[
      \argminOn{\{\Mcoef_{\iatom}\},\,\Mclutter}
        \ltwonorm{\Mimage - \sum_{\iatom} \Mcoef_{\iatom} \Mconv \Matom_{\iatom} - \Mclutter}^{2}
        + \sparsityhp \sum_{\iatom} \lonenorm{\Mcoef_{\iatom}}
        + \lowrankhp \nuclearnorm{\Mclutter}
    \]
  \end{shibuiframe}

  \vspace{0.4em}

  \textbf{Penalty roles:}

  \vspace{0.2em}
  \begin{itemize}
    \setlength{\itemsep}{0.08em}
    \item $\lonenorm{\cdot}$: sparsity of $\Mcoef_{\iatom}$ (few targets)
    \pause
    \item $\nuclearnorm{\cdot}$: low rank of $\Mclutter$ (layers)
    \pause
    \item $\ltwonorm{\cdot}^{2}$: $\ell_2$ fidelity (Gaussian noise)
  \end{itemize}

  \vspace{0.3em}
  \citebadge{Wohlberg (2016); Terrasse (2016)}
\end{frame}

  \textbf{Variable splitting with auxiliary variables:}
  \vspace{0.15em}
  \begin{shibuiframe}{ }
    \small
    Minimize: $\ltwonorm{\Mimage - \sum_{\iatom}\Mcoef_{\iatom}\Mconv\Matom_{\iatom} - \Mclutter}^{2}
    + \sparsityhp\sum_{\iatom}\lonenorm{\Mcoef_{\iatom}} + \lowrankhp\nuclearnorm{\Mclutter}$
  \end{shibuiframe}

  \vspace{0.3em}
  \textbf{Augmented Lagrangian:}
  \vspace{0.15em}
  \begin{shibuiframe}{ }
    \small
    \[
      \lagrangian = \ltwonorm{\Mimage - \sum_{\iatom}\Mcoef_{\iatom}\Mconv\Matom_{\iatom} - \Mclutter}^{2}
         + \sparsityhp\sum_{\iatom}\lonenorm{\Mauxcoef_{\iatom}} + \lowrankhp\nuclearnorm{\Mauxclutter} + \text{(coupling)}
    \]
  \end{shibuiframe}

\end{frame}
