    \column{0.50\textwidth}
      \vspace{0.3em}
      \textbf{Three challenges:}
      \begin{enumerate}
        \item \emphtext{Strong wall returns}\\
              dominate the measurements
        \item \emphtext{Unknown wall properties}\\
              complex dielectric propagation
        \item \emphtext{Multi-path \& low SNR}\\
              for target returns
      \end{enumerate}

      \vspace{0.8em}
      \begin{shibiunote}
        \textbf{Goal:} separate wall returns $\Mwall$
        from target image $\Mtarget$
      \end{shibiunote}
  \end{columns}

  \vspace{0.6em}
  \begin{shibuiwarning}
    \textbf{Unifying theme:} both problems reduce to separating a
    \emphtext{low-rank interference} from \emphtext{sparse target
    responses} -- the same structural challenge.
  \end{shibuiwarning}
\end{frame}
